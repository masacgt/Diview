# CSVインポート画面 Flutter (Dart) 実装仕様書 & フルソースコード

選択された「CSVインポート（アニメーション＆マイクロインタラクション完全版）」のUIデザイン、3ステップウィザード（STEP 1: ファイル選択、STEP 2: 解析中、STEP 3: プレビュー・完了）、各証券会社フォーマット選択（SBI証券、楽天証券、マネックス証券、松井証券、汎用）、CSV出力ガイドのアコーディオン展開、端末内ローカル処理保証バッジを忠実に再現したFlutterコード一式です。

```dart
import 'package:flutter/material.dart';

enum BrokerType {
  sbi,
  rakuten,
  monex,
  matsui,
  custom,
}

class BrokerOption {
  final BrokerType type;
  final String title;
  final String subtitle;
  final IconData icon;
  final bool isRecommended;

  const BrokerOption({
    required this.type,
    required this.title,
    required this.subtitle,
    required this.icon,
    this.isRecommended = false,
  });
}

class BrokerGuideItem {
  final BrokerType type;
  final String name;
  final String description;
  final List<String> steps;

  const BrokerGuideItem({
    required this.type,
    required this.name,
    required this.description,
    required this.steps,
  });
}
```

### B. 証券会社選択グリッド (`widgets/broker_selector_grid.dart`)
```dart
import 'package:flutter/material.dart';
import '../../models/broker_format.dart';
import '../../theme/app_theme.dart';

class BrokerSelectorGrid extends StatelessWidget {
  final List<BrokerOption> brokers;
  final BrokerType selectedBroker;
  final ValueChanged<BrokerType> onSelect;

  const BrokerSelectorGrid({
    super.key,
    required this.brokers,
    required this.selectedBroker,
    required this.onSelect,
  });

  @override
  Widget build(BuildContext context) {
    return Column(
      children: [
        // 上段 2列 (SBI & 楽天)
        Row(
          children: [
            Expanded(child: _buildBrokerTile(brokers[0])),
            const SizedBox(width: 10),
            Expanded(child: _buildBrokerTile(brokers[1])),
          ],
        ),
        const SizedBox(height: 10),
        // 中段 2列 (マネックス & 松井)
        Row(
          children: [
            Expanded(child: _buildBrokerTile(brokers[2])),
            const SizedBox(width: 10),
            Expanded(child: _buildBrokerTile(brokers[3])),
          ],
        ),
        const SizedBox(height: 10),
        // 下段 1列 (汎用)
        _buildBrokerTile(brokers[4], isFullWidth: true),
      ],
    );
  }

  Widget _buildBrokerTile(BrokerOption broker, {bool isFullWidth = false}) {
    final isSelected = selectedBroker == broker.type;

    return InkWell(
      onTap: () => onSelect(broker.type),
      borderRadius: BorderRadius.circular(14),
      child: AnimatedContainer(
        duration: const Duration(milliseconds: 200),
        padding: const EdgeInsets.symmetric(horizontal: 14, vertical: 12),
        decoration: BoxDecoration(
          color: isSelected ? AppColors.surfaceContainerHigh : AppColors.surfaceContainerLow,
          borderRadius: BorderRadius.circular(14),
          border: Border.all(
            color: isSelected ? AppColors.primaryEmerald : AppColors.surfaceContainerHighest.withOpacity(0.6),
            width: isSelected ? 1.5 : 1.0,
          ),
        ),
        child: Row(
          mainAxisAlignment: MainAxisAlignment.spaceBetween,
          children: [
            Row(
              children: [
                Icon(
                  broker.icon,
                  size: 20,
                  color: isSelected ? AppColors.primaryEmerald : AppColors.textSecondary,
                ),
                const SizedBox(width: 10),
                Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text(
                      broker.title,
                      style: TextStyle(
                        fontSize: 13,
                        fontWeight: FontWeight.bold,
                        color: isSelected ? AppColors.textPrimary : AppColors.textSecondary,
                      ),
                    ),
                    Text(
                      broker.subtitle,
                      style: const TextStyle(fontSize: 10, color: AppColors.textTertiary),
                    ),
                  ],
                ),
              ],
            ),
            Container(
              width: 18,
              height: 18,
              decoration: BoxDecoration(
                shape: BoxShape.circle,
                border: Border.all(
                  color: isSelected ? AppColors.primaryEmerald : AppColors.textTertiary,
                  width: 1.5,
                ),
                color: isSelected ? AppColors.primaryEmerald.withOpacity(0.2) : Colors.transparent,

                          color: AppColors.primaryEmerald,
                          shape: BoxShape.circle,
                        ),
                      ),
                    )
                  : null,
            ),
          ],
        ),
      ),
    );
  }
```
