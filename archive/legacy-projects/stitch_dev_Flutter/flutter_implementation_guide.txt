## 3. データモデル (`models/stock_holding.dart`)

```dart
class StockHolding {
  final String code;
  final String name;
  final String market;
  int shares;
  final double currentPrice;
  double costPrice;
  final double dividendYield;
  final double annualDividend;
  final String accountType;
  final String nextPayoutDate;
  final String nextPayoutAmount;
  final bool isUsStock;

  StockHolding({
    required this.code,
    required this.name,
    required this.market,
    required this.shares,
    required this.currentPrice,
    required this.costPrice,
    required this.dividendYield,
    required this.annualDividend,
    required this.accountType,
    required this.nextPayoutDate,
    required this.nextPayoutAmount,
    this.isUsStock = false,
  });

  double get totalValuation => shares * currentPrice;
  double get profitLoss => shares * (currentPrice - costPrice);
  double get profitLossPercent =>
      costPrice > 0 ? ((currentPrice - costPrice) / costPrice) * 100 : 0.0;
}
```
