## 2. データモデル (Models)

```kotlin
package com.kurogane.dividend.model

data class StockHolding(
    val code: String,
    val name: String,
    val market: String,
    val shares: Int,
    val currentPrice: Double,
    val costPrice: Double,
    val dividendYield: Double,
    val annualDividend: Double,
    val accountType: String,
    val nextPayoutDate: String,
    val nextPayoutAmount: String,
    val isUsStock: Boolean = false
) {
    val totalValuation: Double get() = shares * currentPrice
    val profitLoss: Double get() = shares * (currentPrice - costPrice)
    val profitLossPercent: Double get() = if (costPrice > 0) ((currentPrice - costPrice) / costPrice) * 100 else 0.0
}

data class MonthlyDividend(
    val month: Int,
    val amount: Long,
    val isPeakMonth: Boolean = false
)
```

@Composable
fun MonthlyDividendBarSection(totalAnnualDividend: Long) {
    Card(
        shape = RoundedCornerShape(16.dp),
        colors = CardDefaults.cardColors(containerColor = SurfaceContainerLow),
        modifier = Modifier.fillMaxWidth()
    ) {
        Column(modifier = Modifier.padding(16.dp)) {
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                Row(verticalAlignment = Alignment.CenterVertically, horizontalArrangement = Arrangement.spacedBy(6.dp)) {
                    Icon(Icons.Default.CalendarMonth, contentDescription = null, tint = PrimaryEmerald, modifier = Modifier.size(18.dp))
                    Text("月別推移グラフ 2024", fontSize = 14.sp, fontWeight = FontWeight.Bold, color = TextPrimary)
                }
                Text("年間合計: ¥${"%,d".format(totalAnnualDividend)}", fontSize = 12.sp, fontWeight = FontWeight.Bold, color = PrimaryEmerald)
            }

            Spacer(modifier = Modifier.height(14.dp))

            // 12ヶ月バープロット
            val heights = listOf(0.1f, 0.15f, 0.4f, 0.1f, 0.15f, 0.95f, 0.1f, 0.12f, 0.35f, 0.1f, 0.15f, 1.0f)
            Row(
                modifier = Modifier
                    .fillMaxWidth()
                    .height(64.dp),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.Bottom
            ) {
                heights.forEachIndexed { index, fraction ->
                    val isPeak = index == 5 || index == 11
                    Column(horizontalAlignment = Alignment.CenterHorizontally) {
                        Box(
                            modifier = Modifier
                                .width(16.dp)
                                .height((60 * fraction).dp)
                                .clip(RoundedCornerShape(topStart = 4.dp, topEnd = 4.dp))
                                .background(if (isPeak) PrimaryEmerald else SurfaceContainerHighest)
                        )
                        Text("${index + 1}", fontSize = 9.sp, color = if (isPeak) PrimaryEmerald else TextTertiary)
                    }
                }
            }
        }
    }
}

// 5. クイック編集ボトムシート内容
@Composable
fun QuickEditBottomSheetContent(
    stock: StockHolding,
    onDismiss: () -> Unit,
    onSave: (Int, Double) -> Unit
) {
    var shares by remember { mutableStateOf(stock.shares) }
    var costPrice by remember { mutableStateOf(stock.costPrice) }

    Column(
        modifier = Modifier
            .fillMaxWidth()
            .padding(horizontal = 20.dp, vertical = 12.dp)
            .navigationBarsPadding(),
        verticalArrangement = Arrangement.spacedBy(16.dp)
    ) {
        Row(
            modifier = Modifier.fillMaxWidth(),
            horizontalArrangement = Arrangement.SpaceBetween,
            verticalAlignment = Alignment.CenterVertically
        ) {
            Column {
                Text("保有情報のクイック編集", fontSize = 16.sp, fontWeight = FontWeight.Bold, color = TextPrimary)
                Text("${stock.code} ${stock.name}", fontSize = 12.sp, color = PrimaryEmerald)
            }
            TextButton(onClick = onDismiss) {
                Text("閉じる", color = TextSecondary)
            }
        }

        // 数量ステッパー
        Column(verticalArrangement = Arrangement.spacedBy(6.dp)) {
            Text("保有数量 (株)", fontSize = 12.sp, color = TextSecondary)
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                FilledIconButton(
                    onClick = { if (shares > 10) shares -= 10 },
                    colors = IconButtonDefaults.filledIconButtonColors(containerColor = SurfaceContainerHigh)
                ) {
                    Icon(Icons.Default.Remove, contentDescription = null, tint = TextPrimary)
                }
                Text("$shares 株", fontSize = 20.sp, fontWeight = FontWeight.Bold, color = TextPrimary)
                FilledIconButton(
                    onClick = { shares += 10 },
                    colors = IconButtonDefaults.filledIconButtonColors(containerColor = SurfaceContainerHigh)
                ) {
                    Icon(Icons.Default.Add, contentDescription = null, tint = TextPrimary)
                }
            }
        }

        // 取得単価ステッパー
        Column(verticalArrangement = Arrangement.spacedBy(6.dp)) {
            Text("取得単価 (円)", fontSize = 12.sp, color = TextSecondary)
            Row(
                modifier = Modifier.fillMaxWidth(),
                horizontalArrangement = Arrangement.SpaceBetween,
                verticalAlignment = Alignment.CenterVertically
            ) {
                FilledIconButton(
                    onClick = { if (costPrice > 10) costPrice -= 10 },
                    colors = IconButtonDefaults.filledIconButtonColors(containerColor = SurfaceContainerHigh)
                ) {
                    Icon(Icons.Default.Remove, contentDescription = null, tint = TextPrimary)
                }
                Text("¥${"%,.0f".format(costPrice)}", fontSize = 20.sp, fontWeight = FontWeight.Bold, color = TextPrimary)
                FilledIconButton(
                    onClick = { costPrice += 10 },
                    colors = IconButtonDefaults.filledIconButtonColors(containerColor = SurfaceContainerHigh)
                ) {
