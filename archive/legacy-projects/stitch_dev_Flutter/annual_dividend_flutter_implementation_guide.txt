class DividendYearData {
  final int year;
  final int totalAmount;
  final double growthRate; // 前年比 (%)
  final int monthlyAverage;
  final double yoc; // Yield on Cost (簿価利回り %)
  final bool isForecast;
  final String note;

  const DividendYearData({
    required this.year,
    required this.totalAmount,
    required this.growthRate,
    required this.monthlyAverage,
    required this.yoc,
    this.isForecast = false,
    this.note = '',
  });
}

class CompoundingSummary {
  final double fiveYearCagr; // 5年平均増配率 CAGR
  final int cumulativeTotal; // 累計受取総額
  final double yoc; // 簿価利回り
  final int currentAnnualDividend; // 年間予想見込額
  final int monthlyAverage;
  final int yearOverYearIncrease; // 前年比増額

  const CompoundingSummary({
    required this.fiveYearCagr,
    required this.cumulativeTotal,
    required this.yoc,
    required this.currentAnnualDividend,
    required this.monthlyAverage,
    required this.yearOverYearIncrease,
  });
}

class _CompoundChartPainter extends CustomPainter {
  final List<DividendYearData> history;
  final int selectedYear;

  _CompoundChartPainter({
    required this.history,
    required this.selectedYear,
  });

  @override
  void paint(Canvas canvas, Size size) {
    if (history.isEmpty) return;

    final maxVal = 135000.0;
    final bottomY = size.height - 20;
    final chartHeight = bottomY - 10;
    final count = history.length;
    final stepX = size.width / count;

    // グリッド破線
    final gridPaint = Paint()
      ..color = AppColors.surfaceContainerHighest.withOpacity(0.5)
      ..strokeWidth = 1
      ..style = PaintingStyle.stroke;

    for (int i = 1; i <= 3; i++) {
      final y = bottomY - (chartHeight * (i / 3));
      canvas.drawLine(Offset(0, y), Offset(size.width, y), gridPaint);
    }

    final points = <Offset>[];

    // 1. バーの描画
    final barWidth = stepX * 0.42;
    for (int i = 0; i < count; i++) {
      final item = history[i];
      final x = stepX * i + (stepX / 2);
      final heightRatio = item.totalAmount / maxVal;
      final topY = bottomY - (chartHeight * heightRatio);

      points.add(Offset(x, topY));

      final isSelected = item.year == selectedYear;

      // バーのグラデーション
      final rect = RRect.fromRectAndCorners(
        Rect.fromLTRB(x - barWidth / 2, topY, x + barWidth / 2, bottomY),
        topLeft: const Radius.circular(4),
        topRight: const Radius.circular(4),
      );

      final barShader = ui.Gradient.linear(
        Offset(x, topY),
        Offset(x, bottomY),
        [
          isSelected ? AppColors.primaryEmerald : AppColors.primaryContainer.withOpacity(0.8),
          AppColors.primaryContainer.withOpacity(0.2),
        ],
      );

      final barPaint = Paint()
        ..shader = barShader
        ..style = PaintingStyle.fill;

      canvas.drawRRect(rect, barPaint);

      // X軸 年ラベル
      final textSpan = TextSpan(
        text: '${item.year}',
        style: TextStyle(
          color: isSelected ? AppColors.primaryEmerald : AppColors.textTertiary,
          fontSize: 10,
          fontWeight: isSelected ? FontWeight.bold : FontWeight.normal,
        ),
      );
      final textPainter = TextPainter(
        text: textSpan,
        textAlign: TextAlign.center,
        textDirection: TextDirection.ltr,
      )..layout();
      textPainter.paint(canvas, Offset(x - (textPainter.width / 2), bottomY + 4));
    }

    // 2. 複利成長曲線 (CAGR Spline) の描画
    final curvePath = Path();
    curvePath.moveTo(points[0].dx, points[0].dy);

    for (int i = 0; i < points.length - 1; i++) {
      final p0 = points[i];
      final p1 = points[i + 1];
      final controlX1 = p0.dx + (p1.dx - p0.dx) / 2;
      final controlY1 = p0.dy;
      final controlX2 = p0.dx + (p1.dx - p0.dx) / 2;
      final controlY2 = p1.dy;
      curvePath.cubicTo(controlX1, controlY1, controlX2, controlY2, p1.dx, p1.dy);
    }

    final linePaint = Paint()
      ..color = AppColors.cyanAccent
      ..strokeWidth = 2.5
      ..style = PaintingStyle.stroke
      ..strokeCap = StrokeCap.round;

    canvas.drawPath(curvePath, linePaint);

    // 3. ポイントのグロードット描画
    for (int i = 0; i < points.length; i++) {
      final p = points[i];
      final isSelected = history[i].year == selectedYear;

      if (isSelected) {
        // グローエフェクト
        final glowPaint = Paint()
          ..color = AppColors.cyanAccent.withOpacity(0.3)
          ..style = PaintingStyle.fill;
        canvas.drawCircle(p, 8, glowPaint);
      }

      final dotPaint = Paint()
        ..color = isSelected ? Colors.white : AppColors.cyanAccent
        ..style = PaintingStyle.fill;
      canvas.drawCircle(p, isSelected ? 4.5 : 3, dotPaint);
    }
  }

  @override
  bool shouldRepaint(covariant _CompoundChartPainter oldDelegate) {
    return oldDelegate.selectedYear != selectedYear || oldDelegate.history != history;
  }
}

  static String _formatNumber(int number) {
    return number.toString().replaceAllMapped(

    );
  }
